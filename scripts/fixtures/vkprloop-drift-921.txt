def _autofill_skip_reason(n):
    """Latest 'skip #n: ...' line from today's lane-autofill log, if no later 'launch' for #n follows it."""
    import glob
    logs = sorted(glob.glob(D + "/logs/lane-autofill-2*.log"))
    if not logs:
        return ""
    try:
        lines = open(logs[-1]).read().splitlines()[-4000:]
    except OSError:
        return ""
    for ln in reversed(lines):
        if ("launch " in ln and ("-%d " % n in ln or "vkfill-%d-" % n in ln)):
            return ""
        if ("skip #%d:" % n) in ln:
            return ln.split("skip #%d:" % n, 1)[1].strip()
    return ""
